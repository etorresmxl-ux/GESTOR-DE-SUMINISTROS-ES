import streamlit as st

from services.auth_service import authenticate_user


def render(conn):
    """Muestra la pantalla de inicio de sesión y valida contra la tabla usuarios_sistema."""
    st.markdown("## 🔐 Iniciar Sesión - Gestión de Suministro")
    st.write("Por favor, ingresa tus credenciales asignadas para continuar.")

    with st.form("form_login"):
        usuario_input = st.text_input("👤 Usuario:")
        password_input = st.text_input("🔑 Contraseña:", type="password")
        submit_login = st.form_submit_button("🚀 Ingresar", use_container_width=True)

        if submit_login:
            if not usuario_input or not password_input:
                st.warning("⚠️ Por favor ingresa tu usuario y contraseña.")
                return

            try:
                auth_result = authenticate_user(conn, usuario_input, password_input)

                if auth_result:
                    st.session_state["autenticado"] = True
                    st.session_state["usuario_actual"] = auth_result["usuario"]
                    st.session_state["rol_actual"] = auth_result["rol"]
                    st.success("¡Inicio de sesión exitoso! Accediendo...")
                    st.rerun()
                else:
                    st.error("❌ Usuario o contraseña incorrectos. Verifica tus datos.")
            except Exception as e:
                st.error(f"Error al conectar con la base de datos: {e}")
